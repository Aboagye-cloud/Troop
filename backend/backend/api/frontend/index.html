const express = require("express"); const cors = require("cors"); const { GoogleGenAI } = require("@google/genai"); const app = express(); app.use(cors()); app.use(express.json()); // Reads GEMINI_API_KEY set in Vercel environment variables const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY }); app.post("/api/generate", async (req, res) => { const { prompt } = req.body; if (!prompt) { return res.status(400).json({ error: "Prompt is required." }); } try { const response = await ai.models.generateContent({ model: "gemini-2.5-flash", contents: prompt, }); res.json({ result: response.text }); } catch (error) { console.error(error); res.status(500).json({ error: "Failed to generate AI response." }); } }); module.exports = app;
